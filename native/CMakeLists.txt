cmake_minimum_required(VERSION 3.28)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MJXL_BUILD_CLI "Build the command line tool (transitional FFmpeg container backend)" ON)
option(MJXL_BUILD_WINDOWS_ADAPTERS "Build DirectShow and Media Foundation targets" ${WIN32})
option(MJXL_USE_BUNDLED_DEPS "Build pinned libjxl, Brotli and libFLAC sources" ${WIN32})
option(MJXL_BUILD_TESTS "Build portable codec/archive regression checks" OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

if(MJXL_USE_BUNDLED_DEPS)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    foreach(feature TOOLS DEVTOOLS JPEGLI JPEGLI_LIBJPEG DOXYGEN MANPAGES BENCHMARK EXAMPLES JNI SJPEG OPENEXR VIEWERS TCMALLOC PLUGINS)
        set(JPEGXL_ENABLE_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    set(JPEGXL_ENABLE_SKCMS ON CACHE BOOL "" FORCE)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../_deps/libjxl" libjxl EXCLUDE_FROM_ALL)
    foreach(feature BUILD_CXXLIBS BUILD_PROGRAMS BUILD_EXAMPLES BUILD_DOCS WITH_OGG INSTALL_MANPAGES)
        set(${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../_deps/flac" flac EXCLUDE_FROM_ALL)
    if(MSVC)
        # libFLAC 1.5.0 win_utf8_io.c omits declarations for allocation functions.
        target_compile_options(FLAC PRIVATE /FIstdlib.h)
    endif()
    set(MJXL_JXL_LIBS jxl jxl_threads)
    set(MJXL_BROTLI_LIBS brotlienc brotlidec)
    set(MJXL_FLAC_LIBS FLAC)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(JXL REQUIRED IMPORTED_TARGET libjxl libjxl_threads)
    pkg_check_modules(BROTLI REQUIRED IMPORTED_TARGET libbrotlienc libbrotlidec)
    pkg_check_modules(FLAC REQUIRED IMPORTED_TARGET flac)
    set(MJXL_JXL_LIBS PkgConfig::JXL)
    set(MJXL_BROTLI_LIBS PkgConfig::BROTLI)
    set(MJXL_FLAC_LIBS PkgConfig::FLAC)
endif()

if(MSVC)
    add_compile_options(/utf-8 /EHsc /W4)
else()
    add_compile_options(-Wall -Wextra)
endif()
if(WIN32)
    add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
endif()
add_library(mjxl_core STATIC core.cpp)
target_link_libraries(mjxl_core PUBLIC ${MJXL_JXL_LIBS})
add_library(mjxl_audio STATIC audio_policy.cpp flac_audio.cpp)
target_link_libraries(mjxl_audio PRIVATE ${MJXL_FLAC_LIBS})
add_library(mjxl_archive STATIC original_file.cpp)
target_link_libraries(mjxl_archive PRIVATE ${MJXL_BROTLI_LIBS})
foreach(target mjxl_core mjxl_audio mjxl_archive)
    target_include_directories(${target} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
endforeach()

if(MJXL_BUILD_CLI)
    if(WIN32)
        if(NOT FFMPEG_ROOT AND DEFINED ENV{FFMPEG_SDK})
            set(FFMPEG_ROOT "$ENV{FFMPEG_SDK}" CACHE PATH "FFmpeg SDK")
        elseif(NOT FFMPEG_ROOT AND DEFINED ENV{FFMPEG_PATH})
            set(FFMPEG_ROOT "$ENV{FFMPEG_PATH}" CACHE PATH "FFmpeg SDK")
        endif()
        if(NOT FFMPEG_ROOT)
            message(FATAL_ERROR "The transitional container backend requires FFMPEG_ROOT or FFMPEG_SDK")
        endif()
        set(MJXL_CONTAINER_LIBS "${FFMPEG_ROOT}/lib/avformat.lib" "${FFMPEG_ROOT}/lib/avcodec.lib" "${FFMPEG_ROOT}/lib/avutil.lib")
    else()
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(AVFORMAT REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
        set(MJXL_CONTAINER_LIBS PkgConfig::AVFORMAT)
    endif()
    add_library(mjxl_container STATIC container.cpp)
    if(WIN32)
        target_include_directories(mjxl_container SYSTEM PRIVATE "${FFMPEG_ROOT}/include")
    endif()
    target_link_libraries(mjxl_container PRIVATE mjxl_audio ${MJXL_CONTAINER_LIBS})
    target_include_directories(mjxl_container PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(mjxl encoder.cpp cli.cpp)
    target_link_libraries(mjxl PRIVATE mjxl_core mjxl_audio mjxl_archive mjxl_container)
    set_target_properties(mjxl PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../dist")
endif()

if(MJXL_BUILD_TESTS)
    add_executable(mjxl_portability_test portability_test.cpp cli.cpp)
    target_link_libraries(mjxl_portability_test PRIVATE mjxl_audio mjxl_archive)
    add_test(NAME mjxl_portability COMMAND mjxl_portability_test)
endif()

if(MJXL_BUILD_WINDOWS_ADAPTERS)
    if(NOT WIN32)
        message(FATAL_ERROR "Windows adapters require Windows")
    endif()
    set(BASECLASSES "${CMAKE_CURRENT_SOURCE_DIR}/../_deps/windows-samples/Samples/Win7Samples/multimedia/directshow/baseclasses" CACHE PATH "DirectShow base classes")
    file(COPY "${BASECLASSES}/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/dshow_compat" FILES_MATCHING PATTERN "*.cpp" PATTERN "*.h")
    set(BASECLASSES "${CMAKE_CURRENT_BINARY_DIR}/dshow_compat")
    file(READ "${BASECLASSES}/transip.h" COMPAT_TRANSIP)
    string(REPLACE "CTransInPlaceFilter::Copy(" "Copy(" COMPAT_TRANSIP "${COMPAT_TRANSIP}")
    file(WRITE "${BASECLASSES}/transip.h" "${COMPAT_TRANSIP}")
    file(READ "${BASECLASSES}/videoctl.h" COMPAT_VIDEOCTL)
    string(REPLACE "virtual CAggDirectDraw::~CAggDirectDraw()" "virtual ~CAggDirectDraw()" COMPAT_VIDEOCTL "${COMPAT_VIDEOCTL}")
    file(WRITE "${BASECLASSES}/videoctl.h" "${COMPAT_VIDEOCTL}")
    file(GLOB BASE_SOURCES CONFIGURE_DEPENDS "${BASECLASSES}/*.cpp")
    list(FILTER BASE_SOURCES EXCLUDE REGEX "/(dllentry|dllsetup)\\.cpp$")
    add_library(dshow_base STATIC ${BASE_SOURCES})
    target_include_directories(dshow_base PUBLIC "${BASECLASSES}")
    target_compile_options(dshow_base PRIVATE /W0 /wd4996)
    set_target_properties(dshow_base PROPERTIES CXX_STANDARD 14)
    target_link_libraries(dshow_base PUBLIC strmiids winmm msacm32 ole32 oleaut32 uuid advapi32)
    add_library(mjxl_decoder SHARED decoder.cpp decoder.def "${BASECLASSES}/dllentry.cpp" "${BASECLASSES}/dllsetup.cpp")
    set_target_properties(mjxl_decoder PROPERTIES PREFIX "" SUFFIX ".ax")
    target_link_libraries(mjxl_decoder PRIVATE mjxl_core dshow_base)
    add_executable(mjxl_graph_test graph_test.cpp)
    target_link_libraries(mjxl_graph_test PRIVATE strmiids ole32 oleaut32 uuid)
    set_target_properties(mjxl_decoder mjxl_graph_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../dist")
    add_library(mjxl_mft SHARED mft.cpp mft.def)
    target_link_libraries(mjxl_mft PRIVATE mjxl_core mfplat mfuuid ole32 advapi32 runtimeobject)
    set_target_properties(mjxl_mft PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../dist")
    add_executable(mjxl_mft_test mft_test.cpp)
    target_link_libraries(mjxl_mft_test PRIVATE mjxl_core mfplat mfreadwrite mfuuid ole32 runtimeobject)
    set_target_properties(mjxl_mft_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../dist")

endif()
